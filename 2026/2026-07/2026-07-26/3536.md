```cpp
class Solution {
public:
    int maxProduct(int n) {
        int ans = 0;
        int minVal = INT_MIN;
        int otherVal = INT_MIN;
        int newVal;

        while (n > 0) {
            newVal = n%10;
            if (newVal > minVal) {
                minVal = min(otherVal, newVal);
                otherVal = max(otherVal, newVal);
            }
            n /= 10;
        }

        return minVal * otherVal;

    }
};
```

> Time Complexity : `O(log n)`
>
> Space Complexity : `O(1)`
