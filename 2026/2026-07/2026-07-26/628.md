initially didn't think about the two negatives = a positive case.

```cpp
class Solution {
public:
    int maximumProduct(vector<int>& nums) {
        array<int, 5> ans = {INT_MIN, INT_MIN, INT_MIN, INT_MAX, INT_MAX};

        for (int i : nums) {
            if (i > ans[2]) {
                ans[0] = ans[1];
                ans[1] = ans[2];
                ans[2] = i;
            } else if (i > ans[1]) {
                ans[0] = ans[1];
                ans[1] = i;
            } else if (i > ans[0]) {
                ans[0] = i;
            }

            if (i < ans[3]) {
                ans[4] = ans[3];
                ans[3] = i;
            } else if (i < ans[4]) {
                ans[4] = i;
            }
        }

        return max(ans[0] * ans[1] * ans[2], ans[4] * ans[3] * ans[2]);
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
