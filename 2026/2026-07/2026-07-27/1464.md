```cpp
class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int maxi = 0;
        int ans = 0;

        for (int i = 1; i < nums.size(); ++i) {
            ans = max((nums[i] - 1) * (nums[maxi] - 1), ans);
            if (nums[i] > nums[maxi]) {
                maxi = i;
            }
        }

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
