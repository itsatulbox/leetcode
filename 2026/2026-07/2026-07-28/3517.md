```cpp
class Solution {
public:
    string smallestPalindrome(string s) {
        array<int, 26> letters{};
        string mid{};
        string ans{};

        for (char c : s) {
            ++letters[static_cast<int>(c - 'a')];
        } 

        for (int i = 0; i < letters.size(); ++i) {
            if (letters[i] % 2 == 1) {
                mid = static_cast<char>('a' + i);
            }
            for (int j = 0; j < letters[i]/2; ++j) {
                    ans += static_cast<char>('a' + i);
            }
        }

        string rev = ans; 
        reverse(rev.begin(), rev.end());
        ans += mid;
        ans += rev;

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
