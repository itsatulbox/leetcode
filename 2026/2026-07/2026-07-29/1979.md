```cpp
class Solution {
public:
    int findGCD(vector<int>& nums) {
        int minVal{INT_MAX};
        int maxVal{INT_MIN};

        for (int &i : nums) {
            if (i < minVal) {
                minVal = i;
            }
            if (i > maxVal) {
                maxVal = i;
            }
        }

        for (int i = minVal; i > 0; --i) {
            if (!(minVal % i) && !(maxVal % i)) {
                return i;
            }
        }

        return 1;
    }
};
```

> Time Complexity : `O(n + m)`, where `m` is the smallest value in `nums`
>
> Space Complexity : `O(1)`
