didn't fully read the question, and assumed that there could be repeating letters, which cost storage of an array. removed it; have both versions below.

**v1**

```cpp
class Solution {
public:
    int minimumPushes(string word) {
        array<int, 26> alphabet{};
        int ans{};
        for (char &c : word) {
            ++alphabet[static_cast<int>(c - 'a')];
        }

        sort(alphabet.rbegin(), alphabet.rend());

        for (int i = 0; i < alphabet.size(); ++i) {
            ans += alphabet[i] * ((i / 8) + 1);
        }

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`

**v2**

```cpp
class Solution {
public:
    int minimumPushes(string word) {
        int ans{};

        for (int i = 0; i < word.size(); ++i) {
            ans += i / 8 + 1; 
        }

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
