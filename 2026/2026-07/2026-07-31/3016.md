```cpp
class Solution {
public:
    int minimumPushes(string word) {
        array<int, 26> alphabet{};
        int ans{};
        for (char &c : word) {
            ++alphabet[static_cast<int>(c - 'a')];
        }

        sort(alphabet.begin(), alphabet.end(), greater<int>());

        for (int i = 0; i < alphabet.size(); ++i) {
            ans += alphabet[i] * ((i / 8) + 1);
        }

        return ans;
    }
};
```

> Time Complexity : `O(n)`, where `n` is `word.size()`. The sort is over a fixed 26-element array, so it contributes `O(26 log 26)` = `O(1)`
>
> Space Complexity : `O(1)`
