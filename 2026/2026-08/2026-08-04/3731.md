```cpp
class Solution {
public:
    vector<int> findMissingElements(vector<int>& nums) {
        int minVal{INT_MAX};
        int maxVal{INT_MIN};
        array<int, 101> vals{};

        for (int &i : nums) {
            vals[i] = 1;
            minVal = min(i, minVal);
            maxVal = max(i, maxVal);
        }

        vector<int> ans{};
        for (int i = minVal + 1; i < maxVal; ++i) {
            if (!vals[i]) {
                ans.push_back(i);
            }
        }
        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
