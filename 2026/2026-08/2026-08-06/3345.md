```cpp
class Solution {
public:
    int smallestNumber(int n, int t) {
        for (int i = 0; i < t; ++i) {
            if (n + i >= 10) {
                if ((((n + i) % 10) * ((n + i)/10)) % t == 0 ) {
                    return n + i;
                }
            } else {
                if ((n + i) % t == 0) {
                    return n + i;
                }
            }
        }

        return 1;
    }
};
```

> Time Complexity : `O(t)`
>
> Space Complexity : `O(1)`
