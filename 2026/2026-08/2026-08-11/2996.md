title: "Smallest Missing Integer Greater Than Sequential Prefix Sum"
question_id: "2996"

---

```cpp
class Solution {
public:
    int missingInteger(vector<int>& nums) {
        array<int, 51> seen;
        for (int x : nums) {
            seen[x] = 1;
        }

        int sumSeq{nums[0]};
        for (size_t i = 1; i < nums.size() && nums[i] == nums[i - 1] + 1; ++i) {
            sumSeq += nums[i];
        }

        int ans{sumSeq};
        while (ans <= 50 && seen[ans]){
            ++ans;
        } 

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
