```cpp
class Solution {
public:
    int maxSubarrayLength(vector<int>& nums, int k) {
        unordered_map<int, int> freq{}; 

        int l{};
        int r{};
        int count{};

        while (r < nums.size()) {
            if (freq[nums[r]] + 1 <= k) {
                ++freq[nums[r]];
                count = max(count, r - l + 1);
                ++r;
            } else {
                while (l != r) {
                    --freq[nums[l]];
                    ++l;
                    if (nums[l-1] == nums[r]) {
                        break;
                    }
                }
            }
        }
        return count;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(n)`
