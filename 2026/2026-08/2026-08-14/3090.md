title: "Maximum Length Substring With Two Occurrences"
question_id: "3090"

difficulty: "Easy"
---

```cpp
class Solution {
public:
    int maximumLengthSubstring(string s) {
        array<int, 26> alphabet{};

        int l{};
        int ans{};

        for (int r{}; r < s.size(); ++r) {
            if (alphabet[s[r] - 'a'] == 2) {
                while (s[l] != s[r]) {
                    --alphabet[s[l] - 'a'];
                    ++l;
                }
                ++l;
            } else {
                ans = max(ans, r - l + 1);
                ++alphabet[s[r] - 'a'];
            }
        } 

        return ans;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
