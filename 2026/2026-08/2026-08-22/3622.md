```cpp
class Solution {
public:
    bool checkDivisibility(int n) {
        int digSum{};
        int digProd{1}; 
        int nCopy{n};

        while (nCopy > 0) {
            digSum += nCopy % 10;
            digProd *= nCopy % 10;
            nCopy /= 10;
        }
        
        return n % (digProd + digSum) == 0;
    }
};
```

> Time Complexity : `O(n)`
>
> Space Complexity : `O(1)`
