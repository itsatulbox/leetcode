```cpp
class Solution {
public:
    int missingMultiple(vector<int>& nums, int k) {
        bitset<101> vals;
        int ans{k};

        for (int &i : nums) {
            vals.set(i);
        }

        while (ans < 101 && vals[ans]) {
            ans += k;
        }

        return ans;
    }
};
```

> Time Complexity : `O(1)`
>
> Space Complexity : `O(1)`
